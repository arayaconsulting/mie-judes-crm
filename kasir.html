function doGet(e) { if (e.parameter.action == 'getReminders') { return ContentService.createTextOutput(JSON.stringify(getReminders())).setMimeType(ContentService.MimeType.JSON); } return HtmlService.createHtmlOutputFromFile('index'); } function doPost(e) { var action = e.parameter.action; if (action == 'updateTransaksi') { return ContentService.createTextOutput(JSON.stringify(updateTransaksi(e.parameter))).setMimeType(ContentService.MimeType.JSON); } else if (action == 'claimVoucher') { return ContentService.createTextOutput(JSON.stringify(claimVoucher(e.parameter))).setMimeType(ContentService.MimeType.JSON); } else { return ContentService.createTextOutput(JSON.stringify(processForm(e.parameter))).setMimeType(ContentService.MimeType.JSON); } } // 1. FUNGSI PENDAFTARAN MEMBER BARU function processForm(data) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheets()[0]; var kode = "MJ-" + Math.random().toString(36).substr(2, 4).toUpperCase(); sheet.appendRow([new Date(), data.nama.toUpperCase(), "'" + data.hp, data.tgl_lahir, data.pekerjaan, kode, 0, new Date()]); return { nama: data.nama, kode: kode }; } // 2. FUNGSI UPDATE POIN TRANSAKSI function updateTransaksi(data) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheets()[0]; var values = sheet.getDataRange().getValues(); var noHP = data.hp; var nominal = parseFloat(data.nominal); var poinBaru = Math.floor(nominal / 10000); for (var i = 1; i < values.length; i++) { if (values[i][2] == noHP) { var poinLama = Number(values[i][6]) || 0; var totalPoin = poinLama + poinBaru; sheet.getRange(i + 1, 7).setValue(totalPoin); sheet.getRange(i + 1, 8).setValue(new Date()); var pesanWA = "*KONFIRMASI POIN MIE JUDES*\n━━━━━━━━━━━━━━━\nTerima kasih Kak *" + values[i][1] + "*, sudah berkunjung hari ini! 🙏\n\n📝 *DETAIL TRANSAKSI:*\n• Total Belanja: Rp " + nominal.toLocaleString('id-ID') + "\n• Perolehan: +" + poinBaru + " Poin\n\n💰 *SALDO POIN AKHIR: " + totalPoin + " POIN*\n━━━━━━━━━━━━━━━\n\n🎁 *PROGRAM PENUKARAN POIN:* \n🔹 15 Poin : Es Teh Manis 🥤\n🔹 20 Poin : Snack Sosis 🌭\n🔹 25 Poin : Dimsum Special 🥟\n🔹 30 Poin : Mie Kentir / Mie Judes 🍜\n\nKumpulkan terus poinnya! 🔥"; return { success: true, nama: values[i][1], wa: noHP, pesan: pesanWA, totalPoin: totalPoin }; } } return { success: false, message: "Nomor Member tidak terdaftar!" }; } // 3. FUNGSI CEK VOUCHER function claimVoucher(data) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheets()[0]; var values = sheet.getDataRange().getValues(); var kodeInput = data.kode_voucher.toUpperCase().trim(); for (var i = 1; i < values.length; i++) { if (values[i][5] == kodeInput) { return { success: true, nama: values[i][1], wa: values[i][2], pekerjaan: values[i][4], poin: values[i][6], message: "KODE VALID!" }; } } return { success: false, message: "KODE TIDAK DITEMUKAN!" }; } // 4. FUNGSI PENGINGAT (H-1 & HARI H ULTAH + RETENSI 10 HARI) function getReminders() { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheets()[0]; var data = sheet.getDataRange().getValues(); var today = new Date(); // Variabel Besok var besok = new Date(); besok.setDate(today.getDate() + 1); var reminders = []; for (var i = 1; i < data.length; i++) { var nama = data[i][1]; var wa = data[i][2]; var tglLahir = new Date(data[i][3]); var tglKunjungan = new Date(data[i][7]); var saldoPoin = data[i][6]; // CEK HARI H ULANG TAHUN if (tglLahir.getDate() == today.getDate() && tglLahir.getMonth() == today.getMonth()) { reminders.push({ tipe: "ULTAH HARI INI! 🎉", nama: nama, wa: wa, pesan: "*SELAMAT ULANG TAHUN KAK " + nama + "!* 🎂🥳 Spesial di hari bahagia ini, Kakak dapat HADIAH MIE GRATIS dari Mie Judes. Yuk mampir hari ini dan tunjukkan pesan ini ke kasir!" }); } // CEK H-1 ULANG TAHUN else if (tglLahir.getDate() == besok.getDate() && tglLahir.getMonth() == besok.getMonth()) { reminders.push({ tipe: "ULTAH BESOK! 🎁", nama: nama, wa: wa, pesan: "Halo Kak " + nama + ", besok Kakak ulang tahun ya? 😍 Persiapan yuk! Spesial buat besok, ada kado Mie Gratis dari Mie Judes buat Kakak. Sampai jumpa besok ya!" }); } // CEK 10 HARI TIDAK DATANG var selisihHari = Math.floor((today - tglKunjungan) / (1000 * 60 * 60 * 24)); if (selisihHari >= 10) { reminders.push({ tipe: "KANGEN (10 HARI) 🍜", nama: nama, wa: wa, pesan: "Halo Kak " + nama + ", sudah 10 hari nih nggak ketemu di Mie Judes. Saldo poin Kakak masih ada " + saldoPoin + " lho. Mampir jajan mie pedas favorit hari ini yuk!" }); } } return reminders; }
